{{- /* Обработанное фото для галереи. Контекст: dict "src" "alt" "sizes" "eager" "class" Возвращает ; полный WebP кладёт в data-full для лайтбокса. */ -}} {{- $src := .src -}} {{- $alt := .alt | default "" -}} {{- $path := strings.TrimPrefix "/" $src -}} {{- $img := resources.Get $path -}} {{- if and $img (in (slice "jpeg" "png") $img.MediaType.SubType) -}} {{- $full := $img -}} {{- if gt $img.Width 1600 -}} {{- $full = $img.Process "resize 1600x webp q82" -}} {{- else -}} {{- $full = $img.Process "webp q82" -}} {{- end -}} {{- $thumb := $full -}} {{- if gt $img.Width 900 -}} {{- $thumb = $img.Process "resize 900x webp q78" -}} {{- end -}} {{- else -}} {{- end -}}
